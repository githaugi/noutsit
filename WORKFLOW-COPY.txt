name: Build APK
on:
  workflow_dispatch:
  push:
    branches: [main, master]
permissions:
  contents: write
jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 21
      - name: Install dependencies
        run: npm install
      - name: Create Android project
        run: npx cap add android
      - name: Apply icons, camera permission and fixed signing key
        run: |
          cp -r res/. android/app/src/main/res/
          sed -i 's#</manifest>#    <uses-permission android:name="android.permission.CAMERA" />\n    <uses-feature android:name="android.hardware.camera" android:required="false" />\n</manifest>#' android/app/src/main/AndroidManifest.xml
          mkdir -p ~/.android
          cp debug.keystore ~/.android/debug.keystore
      - name: Sync web files
        run: npx cap sync android
      - name: Build APK
        working-directory: android
        run: ./gradlew assembleDebug
      - name: Save build
        uses: actions/upload-artifact@v4
        with:
          name: boards-apk
          path: android/app/build/outputs/apk/debug/app-debug.apk
      - name: Publish release
        uses: softprops/action-gh-release@v2
        with:
          tag_name: build-${{ github.run_number }}
          name: Boards build ${{ github.run_number }}
          files: android/app/build/outputs/apk/debug/app-debug.apk
